<script server>
  export const value = $props.value || ''
export const placeholder = $props.placeholder || ''
export const disabled = $bool($props.disabled)
export const readonly = $bool($props.readonly)
export const size = $props.size || 'md'
export const error = $bool($props.error)
export const label = $props.label || ''
export const helperText = $props.helperText || ''
export const required = $bool($props.required)
export const leftIcon = $props.leftIcon || ''
export const rightIcon = $props.rightIcon || ''
export const clearable = $bool($props.clearable)
export const maxLength = $props.maxLength
export const showCount = $bool($props.showCount)
export const className = $props.className || ''
export const inputClassName = $props.inputClassName || ''

/*
 * Padding decides the height here too, with no `h-*` to override it: an app
 * passing its own `py-1.5` got the padding it asked for and the height it did
 * not, and `className` cannot beat `h-10` at equal specificity
 * (stacksjs/stx#1993).
 *
 * `lg` also needed a padding step adjusted - py-3 around a 1.75rem line-height
 * is 3.25rem of content inside a declared 3rem box - so py-2.5 lands on the
 * 3rem it always claimed. sm and md were already exact.
 */
const sizeClasses = { sm: 'px-3 py-1.5 text-sm', md: 'px-4 py-2 text-base', lg: 'px-5 py-2.5 text-lg' }
const iconSizeClasses = { sm: 'h-4 w-4', md: 'h-5 w-5', lg: 'h-6 w-6' }

const baseClasses = 'block w-full border-0 rounded-control shadow-sm ring-1 ring-inset transition-colors duration-200 focus:ring-2 focus:ring-inset bg-field'
/*
 * The two halves of the validity state, exported so the CLIENT can swap them
 * (stacksjs/stx#2047).
 *
 * `error` was read once on the server, like `disabled` and `loading` used to
 * be -- and validity is the clearest case of state that is not known at render
 * time: a form posted with fetch learns it after the submit, with no
 * re-render. So a field could never BECOME invalid, which made the component a
 * step backwards from a plain
  <input>
  for exactly the forms it is for.
 *
 * Toggled as an object binding rather than a whole class string, because a
 * reactive class string MERGES with the static one -- measured, the element
 * ends up carrying `ring-line-strong` and `ring-danger-line` at once, and
 * which wins is then CSS source order rather than anything this file controls.
 * Per-class toggling removes one set as it adds the other.
 */
export const stateInvalidClasses = 'ring-danger-line text-danger-fg placeholder-danger focus:ring-danger-focus'
export const stateValidClasses = 'ring-line-strong text-fg placeholder-fg-soft focus:ring-accent'
const stateClasses = error ? stateInvalidClasses : stateValidClasses
const disabledClasses = disabled
  ? 'opacity-50 cursor-not-allowed bg-surface'
  : readonly
    ? 'bg-surface cursor-default'
    : 'bg-field'
const paddingClasses = leftIcon ? 'pl-10' : rightIcon || clearable ? 'pr-10' : ''

export const inputClasses = `${baseClasses} ${sizeClasses[size] || sizeClasses.md} ${stateClasses} ${disabledClasses} ${paddingClasses} ${inputClassName}`.trim()
export const labelClasses = `block text-sm font-medium leading-6 mb-2 ${error ? 'text-danger' : 'text-fg'}`.trim()
export const helperClasses = `mt-2 text-sm ${error ? 'text-danger' : 'text-fg-soft'}`.trim()
export const iconClasses = iconSizeClasses[size] || iconSizeClasses.md
/*
 * Ids unique to this instance.
 *
 * These were derived from the LABEL or hardcoded outright, so two of these on
 * one page -- an ordinary form -- emitted the same id twice. `
  <label for>
    `
 * then activated the FIRST field from every label, and `aria-describedby`
 * pointed every field's description at the first one's helper text. A label
 * containing a space also produced an invalid id, `input-First name`, which
 * getElementById and querySelector cannot address at all.
 *
 * `$uid` is unique per instance and stable across renders, which is what an id
 * has to be (utils.ts explains why a render that changes its own bytes is
 * uncacheable). The `id` prop overrides it, for a caller that needs to point
 * its own `for` or `aria-describedby` at this control.
 */
export const inputId = $props.id || `input-${$uid}`
export const helperId = `${inputId}-helper`
/*
 * The attributes that make this a form control (stacksjs/stx#2043).
 *
 * A control with no `name` is not a form control in the HTML sense: it is left
 * out of FormData, out of form.elements.namedItem(), and out of an ordinary
 * non-JS POST. A form built from these components submitted nothing for every
 * field, silently -- which is how six apps read their forms.
 *
 * `autocomplete` is separate and just as load-bearing: it is how the browser
 * and a password manager identify a field, so a password input without
 * `autocomplete="current-password"` is never offered for autofill and is not
 * recognised as part of a sign-in form at all.
 *
 * Emitted only when given, so a control that passes none of them renders
 * exactly the markup it did before.
 */
export const name = $props.name || ''
export const autocomplete = $props.autocomplete || $props.autoComplete || ''
export const form = $props.form || ''
export const inputMode = $props.inputMode || $props.inputmode || ''
export const enterKeyHint = $props.enterKeyHint || $props.enterkeyhint || ''
  </script>
  <script client>
    const emit = defineEmits()
// Validity is live, for the same reason `disabled` is: a form posted with
// fetch learns it after the submit, and nothing re-renders (#2047).
const isError = useReactiveProp('error', {{ error }})
// v-model:value supports modifiers; @input remains available to existing callers.
const inputValue = useModel('value', { default: {{ value }}, parse: (v) => v == null ? '' : String(v) })

function onInput(event) {
  inputValue.input(event.target.value)
  if (!inputValue.modifiers.lazy) emit('input', inputValue())
}

function onChange(event) {
  inputValue.change(event.target.value)
  if (inputValue.modifiers.lazy) emit('input', inputValue())
}

function onClear() {
  inputValue.set('')
  emit('input', '')
  emit('clear')
}

const hasValue = derived(() => String(inputValue()).length > 0)
const charCount = derived(() => `${String(inputValue()).length} / {{ maxLength || 0 }}`)

defineExpose({ value: inputValue, clear: onClear })
  </script>
  <div class="w-full {{ className }}">
    @if(label)
    <label for="{{ inputId }}" class="{{ labelClasses }}">
      {{ label }}
      @if(required)
      <span class="text-danger ml-1">
        *
      </span>
      @endif
    </label>
    @endif
    <div class="relative">
      @if(leftIcon)
      <div class="pointer-events-none absolute inset-y-0 left-0 flex items-center pl-3">
        <span class="{{ iconClasses }} text-fg-soft">
          {!! leftIcon !!}
        </span>
      </div>
      @endif
      <input
      @if(name)name="{{ name }}"@endif
      @if(autocomplete)autocomplete="{{ autocomplete }}"@endif
      @if(form)form="{{ form }}"@endif
      @if(inputMode)inputmode="{{ inputMode }}"@endif
      @if(enterKeyHint)enterkeyhint="{{ enterKeyHint }}"@endif
      type="text"
      id="{{ inputId }}"
      class="{{ inputClasses }}"

      {{-- The served value, alongside the binding that takes over on hydration.
      
           An HTML `value` attribute is the DEFAULT value: it is what the field shows
           on first paint and what an ordinary non-JS POST submits, while the runtime
           drives the live `value` PROPERTY afterwards. With only `:value`, a
           server-rendered `value="a@b.c"` reached the browser as an empty box that
           posted an empty string -- the field arrived under its name carrying
           nothing, which is the same silent shape as the missing name itself
           (stacksjs/stx#2043). The two do not fight: attribute and property are
           different things. --}}
      value="{{ value }}"
      :value="inputValue()"
      placeholder="{{ placeholder }}"
      @if(disabled)disabled@endif
      @if(readonly)readonly@endif
      @if(required)required@endif
      @if(maxLength)maxlength="{{ maxLength }}"@endif
      @input="onInput($event)"
      @change="onChange($event)"
      aria-invalid="{{ error ? 'true' : 'false' }}"
      :aria-invalid="isError() ? 'true' : 'false'"
      :class="{ '{{ stateInvalidClasses }}': isError(), '{{ stateValidClasses }}': !isError() }"
      @if(helperText)aria-describedby="{{ helperId }}"@endif
    />
      @if(clearable)
      <div :show="hasValue() && !{{ disabled }} && !{{ readonly }}" class="absolute inset-y-0 right-0 flex items-center pr-3">
        <button
          type="button"
          @click="onClear()"
          class="text-fg-soft hover:text-fg-strong focus:outline-none"
          aria-label="Clear input"
        >
          <svg class="{{ iconClasses }}" fill="none" viewBox="0 0 24 24" stroke="currentColor">
            <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12" />
          </svg>
        </button>
      </div>
      @endif
    @if(rightIcon && !clearable)
      <div class="pointer-events-none absolute inset-y-0 right-0 flex items-center pr-3">
        <span class="{{ iconClasses }} text-fg-soft">
          {!! rightIcon !!}
        </span>
      </div>
      @endif
    </div>
    @if(helperText || (showCount && maxLength))
    <div class="flex items-center justify-between mt-2">
      @if(helperText)
      <p id="{{ helperId }}" class="{{ helperClasses }}">
        {{ helperText }}
      </p>
      @else
      <span>
      </span>
      @endif

      @if(showCount && maxLength)
      <p class="text-sm text-fg-soft" x-text="charCount()">
      </p>
      @endif
    </div>
    @endif
  </div>
