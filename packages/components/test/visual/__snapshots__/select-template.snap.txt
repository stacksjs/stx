<script server>
  /*
 * `??`, not `||`: a select whose chosen option is the id 0 is ordinary, and
 * `||` served it as nothing selected.
 */
export const value = $props.value ?? ''
/*
 * One shape for every option, whichever way it arrived (stacksjs/stx#2059).
 *
 * `{ value, label, disabled }` objects as before, plus bare strings or numbers
 * (`['S', 'M', 'L']`), which are their own label. A string is parsed as JSON,
 * because that is how a list written as a plain attribute arrives. The client
 * script below applies the same rules to a list that arrives later.
 */
function normaliseOptions(list) {
  if (typeof list === 'string') {
    try { list = JSON.parse(list) }
    catch { list = [] }
  }
  if (!Array.isArray(list))
    return []
  return list.map(option => option !== null && typeof option === 'object'
    ? { value: option.value == null ? '' : String(option.value), label: String(option.label ?? option.value ?? ''), disabled: Boolean(option.disabled) }
    : { value: String(option), label: String(option), disabled: false })
}
export const options = normaliseOptions($props.options)
/*
 * `??`: `placeholder=""` means no placeholder option at all, which a select
 * whose options are its children usually wants. Unset keeps the default.

export const error = $bool($props.error)

export const helperText = $props.helperText || ''

export const className = $props.className || ''

 * plain text, right-aligned, and a 20px disc holding an up and a down
 * chevron. It is still the
  <select>
    below, so the menu it opens, the keys

    <style>

/*
 * Padding decides the height here too, with no `h-*` to override it: an app
 * passing its own `py-1.5` got the padding it asked for and the height it did
 * not, and `className` cannot beat `h-10` at equal specificity
 * (stacksjs/stx#1993).
 *
 * `lg` also needed a padding step adjusted - py-3 around a 1.75rem line-height
 * is 3.25rem of content inside a declared 3rem box - so py-2.5 lands on the
 * 3rem it always claimed. sm and md were already exact.
 */
/*
 * md -- the size nearly every form uses -- reads the density roles, so an
 * app sets its own field size with --stx-control-px, --stx-control-py,
 * --stx-control-text and --stx-control-leading instead of a className
 * that ties with py-2 on specificity. The fallbacks are px-4 / py-2 /
 * text-base, so nothing moves by default (stacksjs/stx#1993).
 */

const baseClasses = 'block w-full border-0 rounded-control shadow-sm ring-1 ring-inset transition-colors duration-200 focus:ring-2 focus:ring-inset bg-field'
/*
 * The two halves of the validity state, exported so the CLIENT can swap them
 * (stacksjs/stx#2047). Toggled per class rather than as a whole string,
 * because a reactive class string MERGES with the static one and the element
 * would carry both rings at once.
 */
export const stateInvalidClasses = 'ring-danger-line text-danger-fg placeholder-danger focus:ring-danger-focus'
export const stateValidClasses = 'ring-line-strong text-fg placeholder-fg-soft focus:ring-accent'

export const helperClasses = `mt-2 text-sm ${error ? 'text-danger' : 'text-fg-soft'}`.trim()
/*
 * Ids unique to this instance.
 *
 * These were derived from the LABEL or hardcoded outright, so two of these on
 * one page -- an ordinary form -- emitted the same id twice. `
      <label for>
        `
 * then activated the FIRST field from every label, and `aria-describedby`
 * pointed every field's description at the first one's helper text. A label
 * containing a space also produced an invalid id, `input-First name`, which
 * getElementById and querySelector cannot address at all.
 *
 * `$uid` is unique per instance and stable across renders, which is what an id
 * has to be (utils.ts explains why a render that changes its own bytes is
 * uncacheable). The `id` prop overrides it, for a caller that needs to point

export const selectId = $props.id || `select-${$uid}`
export const helperId = `${selectId}-helper`

 * A control with no `name` is not a form control in the HTML sense: it is left

 * The options are live (stacksjs/stx#2059).
 *
 * They were read once, in the server script, so a list that arrives with the
 * page's data -- an athlete's groups, a coach's plans -- was not there to read
 * and the select rendered its placeholder alone. Now the prop is a signal and
 * the list below is drawn from it on the client, so a bound `:options` renders
 * and re-renders when the parent's list changes.
 */
const optionSource = useReactiveProp('options', {{ options }}, { parse: v => normaliseOptions(v) })
const optionList = derived(() => normaliseOptions(optionSource()))

/*
 * The choices as a native picker takes them (`items`), for a native host.
 * Disabled choices are left out, since neither platform picker can show one
 * it will not let you pick; the placeholder leads the list only while
 * nothing in it is chosen, as the web's disabled first option does.
 */
const placeholderText = {{ placeholder }}
const nativeItems = derived(() => {
  const list = optionList().filter(option => !option.disabled).map(option => ({ value: option.value, label: option.label }))
  const chosen = list.some(option => option.value === selectedValue())
  return placeholderText && !chosen ? [{ value: '', label: placeholderText }, ...list] : list
})

function onChange(event) {
  selectedValue.set(event.target.value)
  emit('change', selectedValue())
}

defineExpose({ value: selectedValue, options: optionList })
