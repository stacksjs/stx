=== Template ===
<script server>
  export const value = $props.value || ''
export const options = $props.options || []
export const placeholder = $props.placeholder || 'Select an option'
export const disabled = $bool($props.disabled)
export const size = $props.size || 'md'
export const error = $bool($props.error)
export const label = $props.label || ''
export const helperText = $props.helperText || ''
export const required = $bool($props.required)
export const className = $props.className || ''

/*
 * Padding decides the height here too, with no `h-*` to override it: an app
 * passing its own `py-1.5` got the padding it asked for and the height it did
 * not, and `className` cannot beat `h-10` at equal specificity
 * (stacksjs/stx#1993).
 *
 * `lg` also needed a padding step adjusted - py-3 around a 1.75rem line-height
 * is 3.25rem of content inside a declared 3rem box - so py-2.5 lands on the
 * 3rem it always claimed. sm and md were already exact.
 */
const sizeClasses = { sm: 'px-3 py-1.5 text-sm', md: 'px-4 py-2 text-base', lg: 'px-5 py-2.5 text-lg' }
const baseClasses = 'block w-full border-0 rounded-control shadow-sm ring-1 ring-inset transition-colors duration-200 focus:ring-2 focus:ring-inset bg-field'
const stateClasses = error
  ? 'ring-danger-line text-danger-fg placeholder-danger focus:ring-danger-focus'
  : 'ring-line-strong text-fg placeholder-fg-soft focus:ring-accent'
const disabledClasses = disabled
  ? 'opacity-50 cursor-not-allowed bg-surface'
  : 'bg-field cursor-pointer hover:ring-neutral-400 dark:hover:ring-neutral-500'

export const selectClasses = `${baseClasses} ${sizeClasses[size] || sizeClasses.md} ${stateClasses} ${disabledClasses} ${className}`.trim()
export const labelClasses = `block text-sm font-medium leading-6 ${error ? 'text-danger' : 'text-fg'}`.trim()
export const helperClasses = `mt-2 text-sm ${error ? 'text-danger' : 'text-fg-soft'}`.trim()
export const selectId = `select-${value}`
</script>
<script client>
  const emit = defineEmits()
// Reactive prop binding — parent can drive `value` via signals. See #1704.
const selectedValue = useReactiveProp('value', {{ value }}, { parse: (v) => v == null ? '' : String(v) })

function onChange(event) {
  selectedValue.set(event.target.value)
  emit('change', selectedValue())
}

defineExpose({ value: selectedValue })
</script>
<div class="w-full">
  @if(label)
  <label for="{{ selectId }}" class="{{ labelClasses }}">
    {{ label }}
      @if(required)
    <span class="text-danger ml-1">
      *
    </span>
    @endif
  </label>
  @endif
  <div class="{{ label ? 'mt-2' : '' }}">
    <select
      id="{{ selectId }}"
      class="{{ selectClasses }}"
      :value="selectedValue()"
      @if(disabled)disabled@endif
      @if(required)required@endif
      @change="onChange($event)"
      :aria-invalid="{{ error ? 'true' : 'false' }}"
      aria-describedby="{{ helperText ? 'helper-text' : '' }}"
    >
      @if(placeholder)
      <option value="" disabled>
        {{ placeholder }}
      </option>
      @endif

      @foreach(option in options)
      <option
          value="{{ option.value }}"
          @if(option.disabled)disabled@endif
        >
        {{ option.label }}
      </option>
      @endforeach
    </select>
  </div>
  @if(helperText)
  <p id="helper-text" class="{{ helperClasses }}">
    {{ helperText }}
  </p>
  @endif
</div>
