export const disabled = $bool($props.disabled)
export const readonly = $bool($props.readonly)
export const rows = $num($props.rows) || 4
export const autoResize = $bool($props.autoResize)
export const maxRows = $num($props.maxRows) || 10
export const error = $bool($props.error)
export const label = $props.label || ''
/*
 * Ids unique to this instance.
 *
 * These were derived from the LABEL or hardcoded outright, so two of these on
 * one page -- an ordinary form -- emitted the same id twice. `
  <label for>
    `
 * then activated the FIRST field from every label, and `aria-describedby`
 * pointed every field's description at the first one's helper text. A label
 * containing a space also produced an invalid id, `input-First name`, which
 * getElementById and querySelector cannot address at all.
 *
 * `$uid` is unique per instance and stable across renders, which is what an id
 * has to be (utils.ts explains why a render that changes its own bytes is
 * uncacheable). The `id` prop overrides it, for a caller that needs to point
 * its own `for` or `aria-describedby` at this control.
 */
export const inputId = $props.id || `textarea-input-${$uid}`
export const helperId = `${inputId}-helper`
export const helperText = $props.helperText || ''
export const required = $bool($props.required)
export const maxLength = $props.maxLength
export const showCount = $bool($props.showCount)
export const resize = $props.resize || 'vertical'
export const className = $props.className || ''

const resizeClasses = { none: 'resize-none', vertical: 'resize-y', horizontal: 'resize-x', both: 'resize' }
const baseClasses = 'block w-full border-0 rounded-control shadow-sm ring-1 ring-inset transition-colors duration-200 focus:ring-2 focus:ring-inset px-4 py-2 text-base bg-field'
const stateClasses = error
  ? 'ring-danger-line text-danger-fg placeholder-danger focus:ring-danger-focus'
  : 'ring-line-strong text-fg placeholder-fg-soft focus:ring-accent'
const disabledClasses = disabled
  ? 'opacity-50 cursor-not-allowed bg-surface'
  : readonly
    ? 'bg-surface cursor-default'
    : 'bg-field'

export const textareaClasses = `${baseClasses} ${stateClasses} ${disabledClasses} ${resizeClasses[resize] || resizeClasses.vertical} ${className}`.trim()
export const labelClasses = `block text-sm font-medium leading-6 mb-2 ${error ? 'text-danger' : 'text-fg'}`.trim()
export const helperClasses = `mt-2 text-sm ${error ? 'text-danger' : 'text-fg-soft'}`.trim()
  </script>
  <script client>
    const emit = defineEmits()
const autoResize = {{ autoResize }}
const maxRows = {{ maxRows }}
// Reactive prop binding — parent can drive `value` via signals. See #1704.
const inputValue = useReactiveProp('value', {{ value }}, { parse: (v) => v == null ? '' : String(v) })

function resizeTextarea(el) {
  if (!autoResize) return
  el.style.height = 'auto'
  const scrollHeight = el.scrollHeight
  const lineHeight = Number.parseInt(window.getComputedStyle(el).lineHeight) || 20
  const maxHeight = lineHeight * maxRows
  el.style.height = `${Math.min(scrollHeight, maxHeight)}px`
  el.style.overflowY = scrollHeight > maxHeight ? 'auto' : 'hidden'
}

function onInput(event) {
  inputValue.set(event.target.value)
  resizeTextarea(event.target)
  emit('input', inputValue())
}

onMount(() => {
  if (autoResize) {
    const el = document.getElementById('textarea-input')
    if (el) resizeTextarea(el)
  }
})

const charCount = derived(() => `${inputValue().length} / {{ maxLength || 0 }}`)

defineExpose({ value: inputValue })
  </script>
  <div class="w-full">
    @if(label)
    <label for="{{ inputId }}" class="{{ labelClasses }}">
      {{ label }}
      @if(required)
      <span class="text-danger ml-1">
        *
      </span>
      @endif
    </label>
    @endif
    <textarea
    id="{{ inputId }}"
    class="{{ textareaClasses }}"
    rows="{{ rows }}"
    placeholder="{{ placeholder }}"
    @if(disabled)disabled@endif
    @if(readonly)readonly@endif
    @if(required)required@endif
    @if(maxLength)maxlength="{{ maxLength }}"@endif
    @input="onInput($event)"
    :aria-invalid="{{ error ? 'true' : 'false' }}"
    @if(helperText)aria-describedby="{{ helperId }}"@endif
  >
      {{ value }}
    </textarea>
    @if(helperText || (showCount && maxLength))
    <div class="flex items-center justify-between mt-2">
      @if(helperText)
      <p id="{{ helperId }}" class="{{ helperClasses }}">
        {{ helperText }}
      </p>
      @else
      <span>
      </span>
      @endif

      @if(showCount && maxLength)
      <p class="text-sm text-fg-soft" x-text="charCount()">
      </p>
      @endif
    </div>
    @endif
  </div>
