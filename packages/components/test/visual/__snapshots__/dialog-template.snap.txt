<script server>
  export const open = $bool($props.open)
export const className = $props.className || ''

/*
 * The dialog's accessible name.
 *
 * `role="dialog"` with no name is announced as just "dialog", which tells a
 * screen reader user nothing about what has taken over the page -- WCAG 4.1.2.
 * The slot idiom puts the name in a
  <DialogTitle>
    , which the client script
 * below associates on mount; this prop is the server-rendered half, so a
 * dialog is named before hydration and without JavaScript at all. Emitted only
 * when set: an empty aria-label is worse than none, because it suppresses the
 * name the title would otherwise supply. Same idiom as Drawer.
 */
export const title = $props.title || ''

/** Set `closeOnEscape="false"` for a dialog that must be answered, not dismissed. */
export const closeOnEscape = $bool($props.closeOnEscape, true)
/** The same, for a click on the backdrop. */
export const closeOnBackdrop = $bool($props.closeOnBackdrop, true)

const baseClasses = 'relative z-50'
export const dialogClasses = `${baseClasses} ${className}`.trim()

/*
 * This dialog's own id.
 *
 * Every lookup below is scoped to it, so two dialogs on one page - a list and
 * the confirmation for deleting from it - do not take each other's panel or
 * each other's Escape. Same idiom as Drawer, Dropdown and Popover.
 */
export const dialogInstanceId = $uid
  </script>
  <script client>
    const emit = defineEmits()
const dialogInstanceId = {{ dialogInstanceId }}
// useReactiveProp wires `:open="parentSignal()"` from the parent into our
// local `isOpen` signal — when the parent's signal changes, bindAttr
// updates the root's `open` attribute, our MutationObserver picks it up,
// and isOpen flips. See stacksjs/stx#1704.
const isOpen = useReactiveProp('open', {{ open }})
const rootRef = useRef('rootRef')

/*
 * Whether hydration has run.
 *
 * Everything below reaches into the DOM through rootRef, and an effect's first
 * run happens while this setup function is still executing - before x-ref is
 * wired, when the ref can only be null. A dialog rendered open on first paint
 * would otherwise get no scroll lock and no focus. See #1979 for the same
 * mistake in Tabs.
 */
const mounted = state(false)

onMount(() => {
  mounted.set(true)
  labelFromTitle()
})

/*
 * Name the dialog after its
    <DialogTitle>
      , when the server could not.
 *
 * The documented idiom puts the name in the slot, which the server cannot see
 * from here -- it renders the title as opaque slot content. So a dialog
 * without the `title` prop reaches the browser announced as bare "dialog".
 * This associates the two once the slot content exists. Skipped when a name is
 * already present, so the prop always wins.
 */
function labelFromTitle() {
  const root = rootRef.value
  if (!root) return
  if (root.getAttribute('aria-label') || root.getAttribute('aria-labelledby')) return
  const heading = root.querySelector('[data-stx-dialog-title]')
  if (!heading) return
  if (!heading.id) heading.id = `${dialogInstanceId}-title`
  root.setAttribute('aria-labelledby', heading.id)
}

const closeOnEscape = {{ closeOnEscape }}
const closeOnBackdrop = {{ closeOnBackdrop }}

function close() {
  isOpen.set(false)
  emit('close')
}

/** The panel, or the dialog root if this dialog has no DialogPanel in it. */
function panelEl() {
  const root = rootRef.value
  if (!root) return null
  return root.querySelector('[data-stx-dialog-panel]') || root
}

/*
 * A click on the backdrop closes it.
 *
 * The backdrop is a child component, so this listens on the dialog root and
 * asks whether the click landed on the marker DialogBackdrop renders. Checking
 * the target rather than comparing against the panel means a click inside the
 * panel is not a backdrop click even though the backdrop is behind it. Only the
 * dialog on top closes: a click cannot reach one underneath, but a programmatic
 * one could.
 */
function onRootClick(event) {
  if (!closeOnBackdrop || !window.stx.overlays.isTop(dialogInstanceId)) return
  const target = event.target
  if (target && target.closest && target.closest('[data-stx-dialog-backdrop]')) close()
}

/*
 * Open means on the page's overlay stack (window.stx.overlays), which every
 * overlay shares: Dialog, Drawer, stxConfirm and the app's own.
 *
 * It holds the page still from the first overlay to the last, moves focus in
 * here and back to whatever opened it, traps Tab, and gives Escape to the
 * topmost overlay only. This dialog used to do each of those for itself, so
 * with one open over another a single Escape closed both, the two Tab traps
 * fought, and the first to close unlocked the page under the other
 * (stacksjs/stx#2057; the trap, lock and Escape themselves are #1973/#1978).
 */
effect(() => {
  if (!mounted()) return

  if (isOpen()) {
    window.stx.overlays.open(dialogInstanceId, {
      panel: panelEl,
      root: rootRef.value,
      onEscape: closeOnEscape ? close : null,
    })
  }
  else {
    window.stx.overlays.close(dialogInstanceId)
  }
})

onDestroy(() => {
  // Removed while open: off the stack, so the page is not left locked with no
  // dialog on it to unlock it.
  window.stx.overlays.close(dialogInstanceId)
})

defineExpose({
  open: () => isOpen.set(true),
  close,
  isOpen,
})
    </script>
    <div
  x-ref="rootRef"
  :show="isOpen()"
  class="{{ dialogClasses }}"
  data-stx-dialog="{{ dialogInstanceId }}"
  role="dialog"
  aria-modal="true"
  @if(title)aria-label="{{ title }}"@endif
  @click="onRootClick($event)"
>
      <slot />
    </div>
