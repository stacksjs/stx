<script server cache>
  export const variant = $props.variant || 'default'
export const size = $props.size || 'md'
export const dot = $bool($props.dot)
export const removable = $bool($props.removable)

/*
 * Whether this badge announces itself. Off by default (stacksjs/stx#2043-adjacent).
 *
 * Every badge carried `role="status"`, which is an ARIA live region: a screen
 * reader watches it and reads out any change to its contents. A badge is
 * usually a styled label -- a tag, a count, a status pill sitting in a table
 * cell -- and a page of them becomes a page of live regions. One estate has
 * 126 badges on a single view, so a single re-render could queue 126
 * announcements, and the interruptions are the point: a live region speaks
 * over whatever the user is reading.
 *
 * A live region is right when the badge is the thing that CHANGED and nothing
 * else says so -- an unread count going up while focus is elsewhere. That is
 * the caller's judgement, not a default, so it is opt-in: `
  <Badge live>
    ` for
 * polite, `
    <Badge live="assertive">
      ` to interrupt.
 */
const liveProp = $props.live
export const live = liveProp === 'assertive'
  ? 'assertive'
  : ($bool(liveProp) ? 'polite' : '')
export const className = $props.className || ''

/*
 * Role tokens for the status variants, so an app re-themes a badge by setting
 * --stx-accent-soft rather than by cancelling a class (stacksjs/stx#1993).
 * The 100/800 light and 900/200 dark pairing these used IS the soft-fill
 * token's definition, so primary, success, warning and danger are unchanged.
 *
 * `secondary` (purple) and `info` (cyan) stay literal. The role vocabulary has
 * five status hues and neither of those is one of them: mapping info onto the
 * `info` role would paint it blue, which is what `primary` already is, so a
 * badge whose whole job is to distinguish would render two of its variants
 * identically. A wrong colour is worse than an unthemeable one.
 */
const variantClasses = {
  default: 'bg-surface-raised text-fg-strong',
  primary: 'bg-accent-soft text-accent-soft-ink',
  secondary: 'bg-secondary-soft text-secondary-soft-ink',
  success: 'bg-success-soft text-success-soft-ink',
  warning: 'bg-warning-soft text-warning-soft-ink',
  danger: 'bg-danger-soft text-danger-soft-ink',
  info: 'bg-info-soft text-info-soft-ink',
}

const sizeClasses = { sm: 'px-2 py-0.5 text-xs', md: 'px-2.5 py-0.5 text-sm', lg: 'px-3 py-1 text-base' }

const dotColorClasses = {
  default: 'bg-fg-soft',
  primary: 'bg-accent',
  secondary: 'bg-secondary',
  success: 'bg-success',
  warning: 'bg-warning',
  danger: 'bg-danger',
  info: 'bg-info',
}

export const badgeClasses = `inline-flex items-center gap-1.5 font-medium rounded-pill ${variantClasses[variant] || variantClasses.default} ${sizeClasses[size] || sizeClasses.md} ${className}`.trim()
export const dotClasses = `w-2 h-2 rounded-full ${dotColorClasses[variant] || dotColorClasses.default}`.trim()
export const removeButtonClasses = `ml-0.5 -mr-1 text-current hover:bg-black/10 dark:hover:bg-white/10 rounded-full p-0.5 transition-colors duration-150`.trim()
    </script>

function onRemove() {
  emit('remove')
}
    </script>

      </span>

          <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12" />
        </svg>

    </span>
