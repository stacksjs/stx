  export const checked = $bool($props.checked)
export const disabled = $bool($props.disabled)
export const indeterminate = $bool($props.indeterminate)
export const size = $props.size || 'md'
export const label = $props.label || ''
export const description = $props.description || ''
export const error = $bool($props.error)
export const required = $bool($props.required)
export const value = $props.value || ''
export const name = $props.name || ''
export const className = $props.className || ''
// WebKit's native switch retains checkbox form, label and keyboard semantics.
// Other engines keep their ordinary checkbox until they implement `switch`.
export const switchAppearance = $props.appearance === 'switch'
/*
 * `variant="mac"` draws the checkbox a Mac draws: a 14px box with 4px corners,
 * sunk into the page when off and filled with the accent when on. It is the
 * same
  <input type="checkbox">
  , so the form value, the label and the keyboard
 * are the browser's own; only the classes change, to `stx-mac-checkbox`,
 * which this file's
  <style>
    block draws from the `--stx-mac-*` tokens.
 */
export const mac = $props.variant === 'mac'

const sizeClasses = { sm: 'h-4 w-4', md: 'h-5 w-5', lg: 'h-6 w-6' }
const baseClasses = 'rounded border-line-strong text-accent focus:ring-accent focus:ring-2 focus:ring-offset-2 focus:ring-offset-panel transition-colors'
/*
 * The two halves of the validity state, exported so the CLIENT can swap them
 * (stacksjs/stx#2047). Toggled per class rather than as a whole string,
 * because a reactive class string MERGES with the static one and the element
 * would carry both rings at once.
 */
export const stateInvalidClasses = 'border-danger-line text-danger focus:ring-danger-focus'
const stateClasses = error ? stateInvalidClasses : ''
const disabledClasses = disabled ? 'opacity-50 cursor-not-allowed' : 'cursor-pointer'

/*
 * On a native host ($native) the box is the platform's own control, tinted
 * system blue (red while invalid), beside a label at the 17pt body size.
 */
const nativeTint = error ? 'text-danger' : 'text-accent'
export const checkboxClasses = $native
  ? `h-[22px] w-[22px] ${nativeTint} disabled:opacity-50`
  : switchAppearance
    ? `shrink-0 cursor-pointer ${disabledClasses}`.trim()
    : `${baseClasses} ${sizeClasses[size] || sizeClasses.md} ${stateClasses} ${disabledClasses}`.trim()
export const labelClasses = $native
  ? `text-[17px] ${error ? 'text-[#ff3b30] dark:text-[#ff453a]' : 'text-[#000000] dark:text-[#ffffff]'} ${disabled ? 'opacity-50' : ''}`.trim()
  : `text-sm font-medium ${error ? 'text-danger' : 'text-fg'} ${disabled ? 'opacity-50 cursor-not-allowed' : 'cursor-pointer'}`.trim()
export const descriptionClasses = $native
  ? 'text-[15px] text-[#6d6d72] dark:text-[#8d8d93]'
  : `text-sm ${error ? 'text-danger' : 'text-fg-soft'}`.trim()
/*
 * Ids unique to this instance.
 *
 * These were derived from the LABEL or hardcoded outright, so two of these on
 * one page -- an ordinary form -- emitted the same id twice. `
    <label for>
      `
 * then activated the FIRST field from every label, and `aria-describedby`
 * pointed every field's description at the first one's helper text. A label
 * containing a space also produced an invalid id, `input-First name`, which
 * getElementById and querySelector cannot address at all.
 *
 * `$uid` is unique per instance and stable across renders, which is what an id
 * has to be (utils.ts explains why a render that changes its own bytes is
 * uncacheable). The `id` prop overrides it, for a caller that needs to point
 * its own `for` or `aria-describedby` at this control.
 */
export const inputId = $props.id || `checkbox-${$uid}`
export const descriptionId = `${inputId}-description`
/*
 * `autocomplete` and `form`, alongside the `name` this already had
 * (stacksjs/stx#2043). `autocomplete` is how the browser and a password
 * manager identify a field; `form` lets a control live outside the
      <form>
        * element it belongs to. Emitted only when given.
 */
export const autocomplete = $props.autocomplete || $props.autoComplete || ''
export const form = $props.form || ''

      <script client>
        const emit = defineEmits(['change'])
// Validity is live, for the same reason `disabled` is: a form posted with
// fetch learns it after the submit, and nothing re-renders (#2047).
const isError = useReactiveProp('error', {{ error }})
const isDisabled = useReactiveProp('disabled', {{ disabled }})
const isIndeterminate = useReactiveProp('indeterminate', {{ indeterminate }})
// Reactive prop binding so the parent can drive `checked` via signals.
// See stacksjs/stx#1704.
const isChecked = useReactiveProp('checked', {{ checked }})
const currentLabel = useReactiveProp('label', {{ label }}, { parse: value => value ?? '' })

function onInputChange(event) {
  if (isDisabled()) {
    event.target.checked = isChecked()
    return
  }
  isChecked.set(event.target.checked)
  emit('change', isChecked())
}

defineExpose({ isChecked })
