export const indeterminate = $bool($props.indeterminate)
export const size = $props.size || 'md'
export const label = $props.label || ''
export const description = $props.description || ''
export const error = $bool($props.error)
export const required = $bool($props.required)
export const value = $props.value || ''
export const name = $props.name || ''
export const className = $props.className || ''

const sizeClasses = { sm: 'h-4 w-4', md: 'h-5 w-5', lg: 'h-6 w-6' }
const baseClasses = 'rounded border-line-strong text-accent focus:ring-accent focus:ring-2 focus:ring-offset-2 focus:ring-offset-panel transition-colors'
const stateClasses = error ? 'border-danger-line text-danger focus:ring-danger-focus' : ''
const disabledClasses = disabled ? 'opacity-50 cursor-not-allowed' : 'cursor-pointer'

export const checkboxClasses = `${baseClasses} ${sizeClasses[size] || sizeClasses.md} ${stateClasses} ${disabledClasses}`.trim()
export const labelClasses = `text-sm font-medium ${error ? 'text-danger' : 'text-fg'} ${disabled ? 'opacity-50 cursor-not-allowed' : 'cursor-pointer'}`.trim()
export const descriptionClasses = `text-sm ${error ? 'text-danger' : 'text-fg-soft'}`.trim()
/*
 * Ids unique to this instance.
 *
 * These were derived from the LABEL or hardcoded outright, so two of these on
 * one page -- an ordinary form -- emitted the same id twice. `
  <label for>
    `
 * then activated the FIRST field from every label, and `aria-describedby`
 * pointed every field's description at the first one's helper text. A label
 * containing a space also produced an invalid id, `input-First name`, which
 * getElementById and querySelector cannot address at all.
 *
 * `$uid` is unique per instance and stable across renders, which is what an id
 * has to be (utils.ts explains why a render that changes its own bytes is
 * uncacheable). The `id` prop overrides it, for a caller that needs to point
 * its own `for` or `aria-describedby` at this control.
 */
export const inputId = $props.id || `checkbox-${$uid}`
export const descriptionId = `${inputId}-description`
/*
 * `autocomplete` and `form`, alongside the `name` this already had
 * (stacksjs/stx#2043). `autocomplete` is how the browser and a password
 * manager identify a field; `form` lets a control live outside the
    <form>
      * element it belongs to. Emitted only when given.
 */
export const autocomplete = $props.autocomplete || $props.autoComplete || ''
export const form = $props.form || ''
    </script>
    <script client>
      const emit = defineEmits()
const isDisabled = {{ disabled }}
const isIndeterminate = {{ indeterminate }}
// Reactive prop binding so the parent can drive `checked` via signals.
// See stacksjs/stx#1704.
const isChecked = useReactiveProp('checked', {{ checked }})

function onInputChange(event) {
  if (isDisabled) {
    event.target.checked = isChecked()
    return
  }
  isChecked.set(event.target.checked)
  emit('change', isChecked())
}

onMount(() => {
  if (isIndeterminate) {
    const el = document.getElementById({{ inputId }})
    if (el) el.indeterminate = true
  }
})

defineExpose({ isChecked })
    </script>

      @if(form)form="{{ form }}"@endif
      value="{{ value }}"
      :checked="isChecked()"
      @if(disabled)disabled@endif
      @if(required)required@endif
      class="{{ checkboxClasses }}"
      @change="onInputChange($event)"
      :aria-invalid="{{ error ? 'true' : 'false' }}"
      @if(description)aria-describedby="{{ descriptionId }}"@endif
    />
      </div>

        @if(label)

            *
          </span>
          @endif
        </label>
        @endif

          {{ description }}
        </p>
        @endif
      </div>
