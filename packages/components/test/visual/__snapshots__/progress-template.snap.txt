<script server cache>
  export const value = $props.value || 0
export const max = $props.max || 100
export const variant = $props.variant || 'linear'
export const size = $props.size || 'md'
export const color = $props.color || 'primary'
export const showLabel = $bool($props.showLabel)
export const indeterminate = $bool($props.indeterminate)
export const className = $props.className || ''

const linearSizeClasses = { sm: 'h-1', md: 'h-2', lg: 'h-3' }
const circularSizeClasses = { sm: 'w-12 h-12', md: 'w-16 h-16', lg: 'w-24 h-24' }
const colorClasses = {
  primary: 'bg-accent-solid',
  secondary: 'bg-secondary-solid',
  success: 'bg-success-solid',
  warning: 'bg-warning-solid',
  danger: 'bg-danger-solid',
  info: 'bg-cyan-600 dark:bg-cyan-500',
}

export const circularSize = size === 'sm' ? 48 : size === 'md' ? 64 : 96
export const strokeWidth = size === 'sm' ? 4 : size === 'md' ? 6 : 8
export const radius = (circularSize - strokeWidth) / 2
export const circumference = 2 * Math.PI * radius

export const containerClasses = variant === 'linear'
  ? `w-full ${className}`.trim()
  : `inline-flex items-center justify-center ${className}`.trim()

export const linearTrackClasses = `w-full ${linearSizeClasses[size] || linearSizeClasses.md} bg-surface-sunken rounded-full overflow-hidden`.trim()
export const linearBarClasses = `h-full ${colorClasses[color] || colorClasses.primary} transition-all duration-300 ease-in-out ${indeterminate ? 'animate-progress-indeterminate' : ''}`.trim()
export const circularSizeClass = circularSizeClasses[size] || circularSizeClasses.md
export const labelClasses = `text-sm font-medium text-fg-strong mb-2`.trim()
export const circularTrackColor = 'stroke-line'
export const circularBarColor = color === 'primary' ? 'stroke-accent'
  : color === 'secondary' ? 'stroke-secondary-solid'
  : color === 'success' ? 'stroke-success'
  : color === 'warning' ? 'stroke-warning-solid'
  : color === 'danger' ? 'stroke-danger'
  : 'stroke-cyan-600 dark:stroke-cyan-500'
</script>
<script client>
  const circumference = {{ circumference }}
const circularBarColor = {{ circularBarColor }}

/*
 * `value` has been a live prop since #1704. `max` and `indeterminate` were not,
 * and they are bound for the same reasons:
 *
 *   :max="total()"            the denominator is often unknown until the
 *                             response that supplies it arrives
 *   :indeterminate="!total()" and the bar starts indeterminate precisely
 *                             BECAUSE the total is not known yet
 *
 * Frozen, those two could only ever describe the state the page was built in,
 * so a bar could never stop spinning once the total showed up. See #1997 for
 * the same freeze on Button.
 */
/*
 * The locals are deliberately NOT named `max` / `indeterminate`.
 *
 * A client-script name that matches a server-script export stops the server
 * interpolating that name in the template: the `animate-spin` ternary below
 * rendered as literal text rather than being evaluated. The attribute
 * useReactiveProp watches is its first argument, so the local is free to be
 * called something else — and has to be.
 */
const value = useReactiveProp('value', {{ value }})
const maxValue = useReactiveProp('max', {{ max }})
const isIndeterminate = useReactiveProp('indeterminate', {{ indeterminate }})

const percentage = derived(() => Math.min(Math.max((value() / maxValue()) * 100, 0), 100))
const percentLabel = derived(() => `${Math.round(percentage())}%`)
const linearWidth = derived(() => isIndeterminate() ? '30%' : `${percentage()}%`)
const dashOffset = derived(() => isIndeterminate() ? 0 : circumference - (percentage() / 100) * circumference)
// The spin is part of what `indeterminate` MEANS, so it follows the signal
// rather than the server's snapshot of it.
const circularBarClass = derived(() => `${circularBarColor} transition-all duration-300 ease-in-out ${isIndeterminate() ? 'animate-spin' : ''}`.trim())

defineExpose({ value, setValue: (v) => value.set(v) })
</script>
@if(variant === 'linear')
<div class="{{ containerClasses }}">
  @if(showLabel)
  <div class="flex items-center justify-between mb-2">
    <span class="{{ labelClasses }}" x-text="percentLabel()">
    </span>
  </div>
  @endif
  <div
      class="{{ linearTrackClasses }}"
      role="progressbar"
      :aria-valuenow="value()"
      aria-valuemin="0"
      aria-valuemax="{{ max }}"
      :aria-valuemax="maxValue()"
    >
    <div
        class="{{ linearBarClasses }}"
        :style="`width: ${linearWidth()}`"
      >
    </div>
  </div>
</div>
@else
<div class="{{ containerClasses }}">
  <div class="{{ circularSizeClass }} relative">
    <svg
        class="transform -rotate-90"
        width="{{ circularSize }}"
        height="{{ circularSize }}"
        role="progressbar"
        :aria-valuenow="value()"
        aria-valuemin="0"
        aria-valuemax="{{ max }}"
        :aria-valuemax="maxValue()"
      >
      <circle
          class="{{ circularTrackColor }}"
          cx="{{ circularSize / 2 }}"
          cy="{{ circularSize / 2 }}"
          r="{{ radius }}"
          stroke-width="{{ strokeWidth }}"
          fill="none"
        />
      <circle
          class="{{ circularBarColor }} transition-all duration-300 ease-in-out {{ indeterminate ? 'animate-spin' : '' }}"
          :class="circularBarClass()"
          cx="{{ circularSize / 2 }}"
          cy="{{ circularSize / 2 }}"
          r="{{ radius }}"
          stroke-width="{{ strokeWidth }}"
          fill="none"
          stroke-dasharray="{{ circumference }}"
          :stroke-dashoffset="dashOffset()"
          stroke-linecap="round"
        />
    </svg>
    @if(showLabel)
    <div class="absolute inset-0 flex items-center justify-center">
      <span class="text-sm font-semibold text-fg-strong" x-text="percentLabel()">
      </span>
    </div>
    @endif
  </div>
</div>
@endif
<style>
  @keyframes progress-indeterminate {
  0% { transform: translateX(-100%); }
  100% { transform: translateX(400%); }
}

.animate-progress-indeterminate {
  animation: progress-indeterminate 1.5s ease-in-out infinite;
}
</style>
