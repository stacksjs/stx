<script server cache>
  export const content = $props.content || ''
export const position = $props.position || 'top'
export const delay = $num($props.delay) || 0
export const className = $props.className || ''
export const show = $bool($props.show)
export const disabled = $bool($props.disabled)

/*
 * The id that makes the tooltip reachable from its trigger.
 *
 * `role="tooltip"` names what the element IS; it does not attach it to
 * anything. Without `aria-describedby` on the trigger pointing here, a screen
 * reader never reaches this element at all and the role announces nothing --
 * so the component was a DOWNGRADE from the native `title=` it replaces, which
 * at least is announced.
 */
export const tooltipId = `tooltip-${$uid}`

/*
 * Whether the text may wrap. It may, by default.
 *
 * `whitespace-nowrap` keeps a two-word hint on one line, which is what a
 * tooltip usually is -- but a sentence-length one then grows past the viewport
 * instead of wrapping, and on a phone the end of it is simply unreachable.
 * Wrapping with a sensible max width is right for both; `nowrap` is there for
 * a caller who knows the content is short.
 */
export const nowrap = $bool($props.nowrap)

const positionClasses = {
  top: 'bottom-full left-1/2 -translate-x-1/2 mb-2',
  bottom: 'top-full left-1/2 -translate-x-1/2 mt-2',
  left: 'right-full top-1/2 -translate-y-1/2 mr-2',
  right: 'left-full top-1/2 -translate-y-1/2 ml-2',
}

const arrowClasses = {
  top: 'top-full left-1/2 -translate-x-1/2 border-t-inverse',
  bottom: 'bottom-full left-1/2 -translate-x-1/2 border-b-inverse',
  left: 'left-full top-1/2 -translate-y-1/2 border-l-inverse',
  right: 'right-full top-1/2 -translate-y-1/2 border-r-inverse',
}

const wrapClasses = nowrap ? 'whitespace-nowrap' : 'max-w-xs break-words'
/*
 * No `pointer-events-none`: WCAG 2.1 SC 1.4.13 requires content shown on hover
 * to stay visible while the pointer is over it. With the pointer locked out,
 * moving toward the tooltip to read it dismissed it -- worst for exactly the
 * long tooltips that are hardest to read quickly. The panel takes its own
 * mouseenter/mouseleave below so entering it keeps it open.
 */
export const tooltipClasses = `absolute z-50 px-3 py-2 text-sm font-medium text-inverse-ink bg-inverse rounded-panel shadow-lg ${wrapClasses} transition-opacity duration-200 ${positionClasses[position] || positionClasses.top} ${className}`.trim()
export const arrowClass = `absolute w-0 h-0 border-4 border-transparent ${arrowClasses[position] || arrowClasses.top}`.trim()
</script>
<script client>
  const delay = {{ delay }}
// A live prop: a tooltip is commonly suppressed by state the page only learns
// about later — while a field is being edited, or once a hint has been read —
// and frozen it could only ever reflect the state the page was built in.
const tooltipId = {{ tooltipId }}
const triggerRef = useRef('triggerRef')
const isDisabled = useReactiveProp('disabled', {{ disabled }})
// Reactive prop binding — parent can drive `show` via signals. See #1704.
const isVisible = useReactiveProp('show', {{ show }})
let timeoutId = null

function showTooltip() {
  if (isDisabled()) return
  if (delay > 0) {
    timeoutId = setTimeout(() => isVisible.set(true), delay)
  }
  else {
    isVisible.set(true)
  }
}

function hideTooltip() {
  if (timeoutId) {
    clearTimeout(timeoutId)
    timeoutId = null
  }
  isVisible.set(false)
}

/*
 * Escape dismisses it (WCAG 2.1 SC 1.4.13).
 *
 * The rule is that content shown on hover or focus can be dismissed WITHOUT
 * moving the pointer or the focus -- so a keyboard listener on the trigger is
 * not enough when the tooltip was opened by hovering and focus is somewhere
 * else entirely. The listener is global for that reason, and does nothing
 * unless this tooltip is the one showing. useEventListener drops it on
 * teardown, so a page of tooltips does not leak one apiece.
 */
useEventListener('keydown', (event) => {
  if (event.key === 'Escape' && isVisible())
    hideTooltip()
})

/*
 * Put the description on the CONTROL, not on the wrapper around it
 * (stacksjs/stx#2048).
 *
 * `aria-describedby` is not inherited: the description is computed for the
 * element that carries the attribute. On the wrapper that element is a plain,
 * non-focusable
  <div>
    , which assistive technology never lands on -- so the
 * button or input inside the slot got nothing, and an icon-only button lost
 * the name a native `title` would have given it.
 *
 * The events are a different matter and stay on the wrapper: focus and
 * pointer events BUBBLE up from the slotted control, which is why that half
 * worked. A description does not bubble, so it has to be placed.
 *
 * Done on mount because the trigger is the caller's markup, and the component
 * only learns what it is once it exists. Appended rather than assigned: a
 * caller who already described their control keeps that description, and ARIA
 * reads the list in order.
 */
function describeTrigger() {
  const wrapper = triggerRef.value
  if (!wrapper) return
  // The control, or the wrapper itself when the trigger is bare text and
  // there is nothing better to carry it.
  const target = wrapper.firstElementChild || wrapper
  const existing = (target.getAttribute('aria-describedby') || '').split(/\s+/).filter(Boolean)
  if (existing.indexOf(tooltipId) !== -1) return
  existing.push(tooltipId)
  target.setAttribute('aria-describedby', existing.join(' '))
}

onMount(describeTrigger)
  </script>
  <div class="relative inline-block {{ disabled ? 'cursor-not-allowed' : '' }}">
    {{-- The trigger wrapper carries the EVENTS, which bubble up to it from the
       slotted control, and deliberately not the description, which does not
       bubble anywhere. `describeTrigger` below moves that onto the control
       itself (stacksjs/stx#2048). --}}
    <div
    x-ref="triggerRef"
    @mouseenter="showTooltip()"
    @mouseleave="hideTooltip()"
    @focusin="showTooltip()"
    @focusout="hideTooltip()"
  >
      <slot />
    </div>
    <div
    id="{{ tooltipId }}"
    :show="isVisible() && !{{ disabled }}"
    class="{{ tooltipClasses }}"
    role="tooltip"
    @mouseenter="showTooltip()"
    @mouseleave="hideTooltip()"
  >
      {{ content }}
      <div class="{{ arrowClass }}">
      </div>
    </div>
  </div>
