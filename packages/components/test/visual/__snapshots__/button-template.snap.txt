<script server cache>
  export const variant = $props.variant || 'primary'
export const size = $props.size || 'md'
export const disabled = $bool($props.disabled)
export const loading = $bool($props.loading)
export const type = $props.type || 'button'
export const fullWidth = $bool($props.fullWidth)
export const leftIcon = $props.leftIcon || ''
export const rightIcon = $props.rightIcon || ''
export const className = $props.className || ''

const variantClasses = {
  primary: 'bg-blue-500 hover:bg-blue-600 text-white shadow-md hover:shadow-lg dark:bg-blue-600 dark:hover:bg-blue-700',
  secondary: 'bg-surface-sunken hover:bg-neutral-300 text-neutral-900 dark:hover:bg-neutral-600 dark:text-white',
  outline: 'border-2 border-blue-500 text-blue-500 hover:bg-blue-50 dark:border-blue-400 dark:text-blue-400 dark:hover:bg-blue-950',
  ghost: 'hover:bg-surface-raised text-fg-strong',
  danger: 'bg-red-500 hover:bg-red-600 text-white shadow-md hover:shadow-lg dark:bg-red-600 dark:hover:bg-red-700',
}

const sizeClasses = {
  xs: 'px-2.5 py-1 text-xs h-7 gap-1',
  sm: 'px-3 py-1.5 text-sm h-8 gap-1.5',
  md: 'px-4 py-2 text-base h-10 gap-2',
  lg: 'px-6 py-3 text-lg h-12 gap-2.5',
  xl: 'px-8 py-4 text-xl h-14 gap-3',
}

const iconSizeClasses = { xs: 'h-3 w-3', sm: 'h-4 w-4', md: 'h-5 w-5', lg: 'h-6 w-6', xl: 'h-7 w-7' }
const baseClasses = 'inline-flex items-center justify-center font-semibold rounded-md transition-all duration-200 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-blue-500 disabled:opacity-50 disabled:cursor-not-allowed disabled:pointer-events-none'
const widthClasses = fullWidth ? 'w-full' : ''
export const isDisabled = disabled || loading

export const buttonClasses = `${baseClasses} ${variantClasses[variant] || variantClasses.primary} ${sizeClasses[size] || sizeClasses.md} ${widthClasses} ${className}`.trim()
export const iconClasses = iconSizeClasses[size] || iconSizeClasses.md
export const spinnerClasses = `animate-spin ${iconClasses}`.trim()
</script>
<script client>
  const emit = defineEmits()

/*
 * `disabled` and `loading` are read as live props, not as a server snapshot.
 *
 * This used to be `const disabledNow = {{ isDisabled }}` — a value computed
 * before the page was sent and never looked at again. So the most common piece
 * of button state in an application could not be expressed:
 *
 *
  <Button :disabled="saving()" @click="save">
    Save
  </Button>
  *
 * `saving()` is a signal. On the server it is not yet meaningful, so the button
 * shipped enabled and STAYED enabled for the life of the page — through the
 * exact async action it was there to guard. Double submits, destructive
 * deletes and rotate-key confirmations all land on this.
 *
 * useReactiveProp wires the parent's binding into a local signal: the parent's
 * bindAttr writes the attribute on our root, a MutationObserver picks it up,
 * and the signal flips. Dialog already does this for `open` (#1704) and Switch
 * for `checked`, so this is the package's existing answer rather than a new one.
 *
 * See stacksjs/stx#1997.
 */
const isLoading = useReactiveProp('loading', {{ loading }})
const disabledProp = useReactiveProp('disabled', {{ disabled }})

// Loading implies disabled, the same combination the server computes for the
// initial render, so the two agree before and after hydration.
const isDisabled = derived(() => disabledProp() || isLoading())

function onClick(event) {
  // Read through the signal: a button disabled after render must stop guarding
  // on the live value, not on what was true when the page was built.
  if (isDisabled()) {
    event.preventDefault()
    return
  }
  emit('click', event)
}
</script>
<button
  type="{{ type }}"
  class="{{ buttonClasses }}"
  @if(isDisabled)disabled@endif
  :disabled="isDisabled()"
  aria-busy="{{ loading ? 'true' : 'false' }}"
  :aria-busy="isLoading() ? 'true' : 'false'"
  @click="onClick($event)"
>
  {{-- The spinner is always in the DOM so `loading` can turn it on after render;
       it was inside an @if, which meant the only way to see it was to have been
       loading when the page was built. x-cloak only when the server says we are
       NOT loading, so a button rendered loading shows its spinner on first
       paint rather than waiting for hydration. --}}
  <svg
    :show="isLoading()"
    @if(!loading)x-cloak@endif
    class="{{ spinnerClasses }}"
    xmlns="http://www.w3.org/2000/svg"
    fill="none"
    viewBox="0 0 24 24"
  >
    <circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4">
    </circle>
    <path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z">
    </path>
  </svg>
  @if(leftIcon)
  <span :show="!isLoading()" @if(loading)x-cloak@endif class="{{ iconClasses }}">
    {!! leftIcon !!}
  </span>
  @endif
  <slot />
  @if(rightIcon)
  <span :show="!isLoading()" @if(loading)x-cloak@endif class="{{ iconClasses }}">
    {!! rightIcon !!}
  </span>
  @endif
</button>
